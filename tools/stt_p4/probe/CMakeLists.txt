cmake_minimum_required(VERSION 3.16)

# This standalone project deliberately reuses inspected local components. It
# never resolves/downloads replacements and never supplies a new flash layout.
foreach(required ESPDL_COMPONENT_DIR P4_PARTITION_CSV STT_MODEL_DIR STT_FIXTURE_DIR)
    if(NOT DEFINED ${required} OR NOT IS_ABSOLUTE "${${required}}" OR NOT EXISTS "${${required}}")
        message(FATAL_ERROR "Supply -D${required}=<existing absolute path>")
    endif()
    if("${${required}}" MATCHES "[;\n\r\"]")
        message(FATAL_ERROR "Unsupported character in ${required}")
    endif()
endforeach()
get_filename_component(_managed "${ESPDL_COMPONENT_DIR}" DIRECTORY)
set(_fft "${_managed}/espressif__dl_fft")
set(_jpeg "${_managed}/espressif__esp_new_jpeg")
function(require_component_version dir expected)
    if(NOT EXISTS "${dir}/idf_component.yml")
        message(FATAL_ERROR "Missing inspected component ${dir}")
    endif()
    file(STRINGS "${dir}/idf_component.yml" _versions REGEX "^version: ")
    if(NOT _versions STREQUAL "version: ${expected}")
        message(FATAL_ERROR "Expected component version ${expected}: ${dir}; found ${_versions}")
    endif()
endfunction()
require_component_version("${ESPDL_COMPONENT_DIR}" "3.3.12")
require_component_version("${_fft}" "0.7.0")
require_component_version("${_jpeg}" "1.0.2")
set(ENV{IDF_COMPONENT_MANAGER} "0")
# Without component-manager namespace rewriting, the component CMake files
# require the plain names dl_fft and esp_new_jpeg. Build-local symlinks provide
# those names while compiling the original inspected source directories.
set(_links "${CMAKE_BINARY_DIR}/local-components")
file(MAKE_DIRECTORY "${_links}")
function(link_local_component name source)
    set(destination "${_links}/${name}")
    if(IS_SYMLINK "${destination}")
        file(READ_SYMLINK "${destination}" existing)
        if(NOT existing STREQUAL source)
            message(FATAL_ERROR "Stale component link ${destination}; use a fresh build directory")
        endif()
    elseif(EXISTS "${destination}")
        message(FATAL_ERROR "Unexpected existing component ${destination}")
    else()
        file(CREATE_LINK "${source}" "${destination}" SYMBOLIC RESULT result)
        if(NOT result STREQUAL "0")
            message(FATAL_ERROR "Cannot link local component ${name}: ${result}")
        endif()
    endif()
endfunction()
link_local_component("esp-dl" "${ESPDL_COMPONENT_DIR}")
link_local_component("dl_fft" "${_fft}")
link_local_component("esp_new_jpeg" "${_jpeg}")
set(EXTRA_COMPONENT_DIRS "${_links}/esp-dl" "${_links}/dl_fft" "${_links}/esp_new_jpeg")
link_local_component("littlefs" "${_managed}/joltwallet__littlefs")
list(APPEND EXTRA_COMPONENT_DIRS "${_links}/littlefs")
set(COMPONENTS main)
set(IDF_TARGET esp32p4 CACHE STRING "Only the P4 operator probe is supported" FORCE)

# Copy the supplied layout verbatim into the build directory. Configure fails
# later if a reused sdkconfig tries to select another partition layout.
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}")
configure_file("${P4_PARTITION_CSV}" "${CMAKE_BINARY_DIR}/probe-partitions.csv" COPYONLY)
set(_partition_defaults "${CMAKE_BINARY_DIR}/partition.defaults")
file(WRITE "${_partition_defaults}"
    "CONFIG_PARTITION_TABLE_CUSTOM=y\nCONFIG_PARTITION_TABLE_CUSTOM_FILENAME=\"${CMAKE_BINARY_DIR}/probe-partitions.csv\"\nCONFIG_PARTITION_TABLE_OFFSET=0x8000\n")
set(SDKCONFIG_DEFAULTS "${CMAKE_CURRENT_LIST_DIR}/sdkconfig.defaults;${_partition_defaults}")
if(NOT DEFINED SDKCONFIG)
    set(SDKCONFIG "${CMAKE_BINARY_DIR}/sdkconfig")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(hw1_p4_stt_probe)
if(NOT IDF_VERSION_MAJOR EQUAL 5 OR NOT IDF_VERSION_MINOR EQUAL 5 OR NOT IDF_VERSION_PATCH EQUAL 5)
    message(FATAL_ERROR "Use the inspected ESP-IDF 5.5.5 checkout")
endif()
if(NOT CONFIG_PARTITION_TABLE_CUSTOM OR NOT CONFIG_PARTITION_TABLE_OFFSET EQUAL 32768)
    message(FATAL_ERROR "Expected the existing P4 custom partition table at 0x8000")
endif()
if(NOT CONFIG_PARTITION_TABLE_CUSTOM_FILENAME STREQUAL "${CMAKE_BINARY_DIR}/probe-partitions.csv")
    message(FATAL_ERROR "Stale sdkconfig selected another partition file; use a fresh build directory")
endif()
file(SHA256 "${P4_PARTITION_CSV}" _partition_sha)
message(STATUS "STT probe supplied partition CSV SHA256: ${_partition_sha}")
